    }

    function renderMainPagination() {
      const lastPage = Math.max(1, Math.ceil(currentMatches.length / PAGE_SIZE));
      if (currentPage > lastPage) currentPage = lastPage;
      if (currentPage < 1) currentPage = 1;

      // Cache les cartes hors page
      const cards = allArticlesGrid.querySelectorAll('.article-card');
      cards.forEach((card) => {
        const p = parseInt(card.getAttribute('data-page') || '1', 10);
        card.hidden = p !== currentPage;
      });

      renderPagination(lastPage);
      paginationStatus.textContent = 'Page ' + currentPage + ' sur ' + lastPage + ' (' + currentMatches.length + ' article' + (currentMatches.length > 1 ? 's' : '') + ')';
    }

    function goToPage(p) {
      currentPage = p;
      renderMainPagination();
      // Scroll vers la grille (pas tout en haut de page)
      const top = allArticlesGrid.getBoundingClientRect().top + window.scrollY - 120;
      window.scrollTo({ top, behavior: 'smooth' });
    }

    // Bouton "Voir plus" pour les tags.
    const tagToggle = document.getElementById('tag-toggle');
    const tagOverflow = document.getElementById('tag-overflow');
    if (tagToggle && tagOverflow) {
      tagToggle.addEventListener('click', () => {
        const expanded = tagToggle.getAttribute('aria-expanded') === 'true';
        const next = !expanded;
        tagToggle.setAttribute('aria-expanded', String(next));
        tagOverflow.hidden = !next;
      });
    }

    // Filtre par tag (pill).
    tagButtons.forEach((btn) => {
      btn.addEventListener('click', (e) => {
        // Liens : on laisse la navigation se faire naturellement (vers /blog/tag/...).
        if (btn.tagName === 'A') return;
        e.preventDefault();
        tagButtons.forEach((b) => b.classList.remove('active'));
        btn.classList.add('active');
        activeTag = btn.getAttribute('data-tag') || 'all';
        currentPage = 1;
        applyFilter();
        if (activeTag !== 'all') {
          document.getElementById('search-results')?.scrollIntoView({ behavior: 'smooth', block: 'start' });
        }
      });
    });

    // Recherche live (debounced).
    let searchTimer = null;
    searchInput.addEventListener('input', () => {
      window.clearTimeout(searchTimer);
      searchTimer = window.setTimeout(() => {
        if (searchInput.value.trim()) {
          searchClear.hidden = false;
        } else {
          searchClear.hidden = true;
        }
        applyFilter();
      }, 120);
    });

    searchClear.addEventListener('click', () => {
      searchInput.value = '';
      searchClear.hidden = true;
      applyFilter();
      searchInput.focus();
    });

    // Boutons de pagination.
    paginationPrev.addEventListener('click', () => {
      if (currentPage > 1) goToPage(currentPage - 1);
    });
    paginationNext.addEventListener('click', () => {
      const lastPage = Math.max(1, Math.ceil(currentMatches.length / PAGE_SIZE));
      if (currentPage < lastPage) goToPage(currentPage + 1);
    });
    paginationPages.addEventListener('click', (e) => {
      const t = e.target;
      if (!(t instanceof HTMLElement)) return;
      if (t.tagName !== 'BUTTON') return;
      const p = parseInt(t.getAttribute('data-page') || '1', 10);
      if (!isNaN(p)) goToPage(p);
    });

    // Raccourci "/" pour focaliser la recherche.
    document.addEventListener('keydown', (e) => {
      if (e.key === '/' && document.activeElement !== searchInput && document.activeElement?.tagName !== 'INPUT' && document.activeElement?.tagName !== 'TEXTAREA') {
        e.preventDefault();
        searchInput.focus();
        searchInput.select();
      }
      if (e.key === 'Escape' && document.activeElement === searchInput) {
        searchInput.value = '';
        searchClear.hidden = true;
        applyFilter();
        searchInput.blur();
      }
    });

    // Initialisation : afficher la page 1
    renderMainPagination();
  })();
})();</script>
